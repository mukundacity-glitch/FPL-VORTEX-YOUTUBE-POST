    ASSEMBLER_VERSION = "VX21-PRODUCTION-PACKAGING-V10-NARRATION-PACING"

    def _vx21_build_transition_overlay(next_label, preview_png, transition_number):
        """Build a transparent premium lower-third + live next-slide preview overlay."""
        from PIL import Image as _Img, ImageDraw as _Draw, ImageFont as _Font, ImageFilter as _Filter, ImageOps as _Ops

        preview_png = Path(preview_png)
        if not preview_png.is_file() or preview_png.stat().st_size <= 0:
            raise FileNotFoundError(f"Transition preview is missing: {preview_png}")

        overlay = _Img.new("RGBA", (int(W), int(H)), (0, 0, 0, 0))
        draw = _Draw.Draw(overlay)
        width, height = overlay.size

        # Cinematic lower-third glass band, deliberately below the centre.
        band_top = int(height * 0.645)
        band_bottom = int(height * 0.955)
        band = _Img.new("RGBA", (width, band_bottom - band_top), (0, 0, 0, 0))
        bp = band.load()
        for yy in range(band.height):
            frac = yy / max(1, band.height - 1)
            alpha = int(78 + 112 * frac)
            for xx in range(width):
                bp[xx, yy] = (3, 14, 27, alpha)
        overlay.alpha_composite(band, (0, band_top))

        # Accent geometry.
        left_x = int(width * 0.052)
        accent_y = int(height * 0.705)
        draw.rounded_rectangle((left_x, accent_y, left_x + int(width * 0.010), int(height * 0.905)),
                               radius=max(4, int(height * .006)), fill=(51, 232, 255, 238))
        draw.rounded_rectangle((left_x + int(width * .014), accent_y, left_x + int(width * .020), int(height * .850)),
                               radius=max(3, int(height * .004)), fill=(0, 255, 133, 210))

        # Colab images do not guarantee a specific system-font package. Prefer the
        # premium fonts already cached by Cell 7A, then fall back safely.
        _font_cache = Path(
            globals().get("HTML_WORK_DIR", "/content/FPL_VORTEX_DAY3_WORK/html")
        ).parent / "premium_fonts"
        _font_candidates_bold = (
            _font_cache / "BarlowCondensed-Bold.ttf",
            _font_cache / "Montserrat[wght].ttf",
            Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"),
            Path("/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf"),
        )
        _font_candidates_regular = (
            _font_cache / "BarlowCondensed-Regular.ttf",
            _font_cache / "Montserrat[wght].ttf",
            Path("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"),
            Path("/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf"),
        )

        def _vx21_load_overlay_font(size, *, bold=False):
            candidates = _font_candidates_bold if bold else _font_candidates_regular
            for candidate in candidates:
                if not candidate.is_file():
                    continue
                try:
                    return _Font.truetype(str(candidate), max(10, int(size)))
                except (OSError, ValueError):
                    continue
            try:
                return _Font.load_default(size=max(10, int(size)))
            except TypeError:
                return _Font.load_default()

        label_font = _vx21_load_overlay_font(max(22, int(height * .026)), bold=True)
        draw.text((left_x + int(width*.036), int(height*.695)), "UP NEXT", font=label_font, fill=(83, 235, 255, 245),
                  stroke_width=max(1, int(height*.0008)), stroke_fill=(0, 20, 35, 220))

        # Fit the main heading into no more than two lines in the lower-left zone.
        title_x = left_x + int(width * .036)
        title_y = int(height * .755)
        max_title_width = int(width * .535)
        words = str(next_label).replace("•", " ").split()
        chosen_font = None
        chosen_lines = None
        for size in range(max(42, int(height*.062)), max(30, int(height*.036)) - 1, -2):
            fnt = _vx21_load_overlay_font(size, bold=True)
            lines=[]; cur=""
            for word in words:
                trial=(cur+" "+word).strip()
                box=draw.textbbox((0,0), trial, font=fnt, stroke_width=max(1,int(size*.025)))
                if cur and (box[2]-box[0]) > max_title_width:
                    lines.append(cur); cur=word
                else:
                    cur=trial
            if cur: lines.append(cur)
            if len(lines) <= 2:
                chosen_font=fnt; chosen_lines=lines; break
        if chosen_font is None:
            chosen_font=_vx21_load_overlay_font(max(30,int(height*.036)), bold=True)
            chosen_lines=[str(next_label)]

        # Gradient-filled title mask for an artistic broadcast look.
        line_gap = int(height * .008)
        stroke = max(2, int(height * .0016))
        line_heights=[]
        max_w=1
        for line in chosen_lines:
            b=draw.textbbox((0,0), line, font=chosen_font, stroke_width=stroke)
            line_heights.append(b[3]-b[1]); max_w=max(max_w,b[2]-b[0])
        total_h=sum(line_heights)+line_gap*max(0,len(chosen_lines)-1)
        mask=_Img.new("L", (max_w+stroke*6,total_h+stroke*6), 0)
        md=_Draw.Draw(mask); yy=stroke*2
        for li,line in enumerate(chosen_lines):
            md.text((stroke*2,yy), line, font=chosen_font, fill=255, stroke_width=stroke, stroke_fill=255)
            yy += line_heights[li] + line_gap
        grad=_Img.new("RGBA", mask.size, (0,0,0,0)); gp=grad.load()
        stops=((0.0,(52,226,255,255)),(.48,(0,255,133,255)),(1.0,(205,85,255,255)))
        for xx in range(grad.width):
            t=xx/max(1,grad.width-1)
            if t <= stops[1][0]:
                p=t/stops[1][0]; a=stops[0][1]; b=stops[1][1]
            else:
                p=(t-stops[1][0])/(1-stops[1][0]); a=stops[1][1]; b=stops[2][1]
            col=tuple(int(a[i]+(b[i]-a[i])*p) for i in range(4))
            for yy2 in range(grad.height): gp[xx,yy2]=col
        title_layer=_Img.new("RGBA", overlay.size, (0,0,0,0))
        title_layer.paste(grad, (title_x, title_y), mask)
        overlay = _Img.alpha_composite(overlay, title_layer)
        draw = _Draw.Draw(overlay)

        # Live preview of the actual next slide, not a hard-coded illustration.
        card_w = int(width * .315)
        card_h = int(height * .315)
        card_x = int(width * .646)
        card_y = int(height * .620)
        shadow = _Img.new("RGBA", overlay.size, (0,0,0,0))
        sd = _Draw.Draw(shadow)
        radius=max(18,int(height*.018))
        sd.rounded_rectangle((card_x+int(width*.006), card_y+int(height*.012), card_x+card_w+int(width*.006), card_y+card_h+int(height*.012)),
                             radius=radius, fill=(0,0,0,155))
        shadow=shadow.filter(_Filter.GaussianBlur(max(8,int(height*.010))))
        overlay=_Img.alpha_composite(overlay,shadow)

        preview=_Img.open(preview_png).convert("RGB")
        inner_w=card_w-int(width*.012); inner_h=card_h-int(height*.022)
        fitted=_Ops.fit(preview,(inner_w,inner_h),method=_Img.Resampling.LANCZOS,centering=(.5,.5))
        mask_card=_Img.new("L", fitted.size,0); mcd=_Draw.Draw(mask_card)
        mcd.rounded_rectangle((0,0,fitted.width-1,fitted.height-1), radius=max(14,int(height*.014)), fill=255)
        preview_rgba=_Img.new("RGBA", fitted.size,(0,0,0,0)); preview_rgba.paste(fitted,(0,0),mask_card)
        px=card_x+int(width*.006); py=card_y+int(height*.011)
        overlay.alpha_composite(preview_rgba,(px,py))
        draw=_Draw.Draw(overlay)
        draw.rounded_rectangle((card_x,card_y,card_x+card_w,card_y+card_h), radius=radius, outline=(62,226,255,240), width=max(3,int(height*.0022)))
        draw.rounded_rectangle((card_x+max(6,int(width*.004)),card_y+max(6,int(height*.006)),card_x+card_w-max(6,int(width*.004)),card_y+card_h-max(6,int(height*.006))),
                               radius=max(12,radius-5), outline=(0,255,133,120), width=max(2,int(height*.0012)))

        out = ASSEMBLY_WORK / f"transition_overlay_{int(transition_number):02d}.png"
        overlay.save(out, format="PNG", optimize=True)
        if not out.is_file() or out.stat().st_size <= 0:
            raise RuntimeError(f"Failed to create transition overlay: {out}")
        return out
